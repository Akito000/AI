using System.Collections.Generic;
using System.Linq;
using UnityEngine;
using Game.CharacterInput;
using Game.World;

namespace Game.AI
{
    /// <summary>
    /// Attacker用「同色の敵スポナーを破壊」。UtilityAIとゴールベースAIのハイブリッドとして扱う特殊処理。
    ///
    /// 通常のこのゲームのUtility AIは「毎Think、全Action×全Targetを独立に再比較する」設計のため、
    /// 従来のようにMoveToSpawnerAction(移動)とDestroySpawnerAction(攻撃)へ分けてしまうと、
    /// 間合いの境界付近でスコアがわずかに揺れるだけで「近づく⇔攻撃する」を行き来したり、
    /// 別のスポナーのスコアがたまたま上回った瞬間に対象を乗り換えたりして、
    /// 「近づいて、着いたら殴る」という一連の流れが崩れやすい。
    ///
    /// そこでこのActionでは「今どのスポナーを狙うか」の初期選定だけをUtility AI(Considerationsによる
    /// スコア比較)に任せ、一度選んだらそのスポナーが無効(破壊された/対象外になった)になるまでは
    /// GetTargetCandidatesがそのTargetだけを返し続ける「ゴール」として扱う(ゴールベースAI的な固定)。
    /// ゴールを持っている間のAct()内部では、間合い外なら接近・間合い内なら攻撃という単純な2状態の
    /// ミニステートマシンを直接実行し、Utility Considerationsの再比較を挟まない。
    /// 危険な攻撃が迫った場合等は、他のIsUrgentCapable=trueなActionがUrgent Checkで上書きできるため、
    /// ゴールに固執して安全確認を無視するわけではない。
    /// </summary>
    public class DestroySpawnerGoalAction : IUtilityAction, IMotionLockingAction
    {
        public string Name => "DestroySpawnerGoal";
        public float BaseWeight => 1.0f;
        public bool IsUrgentCapable => false;

        private const float CooldownDuration = 1.2f;
        private const float DamagePerHit = 10f;
        private const float AttackMotionLockSeconds = 0.35f;

        /// <summary>現在確定しているゴール(狙っているスポナー)。無効化されるまで他候補を見ない。</summary>
        private Transform _committedSpawner;
        private bool _isAttackPhase;

        /// <summary>攻撃フェーズの間だけ専有する(接近フェーズはThinkを毎回再評価させ、経路を追従させたい)。</summary>
        public float MotionLockSeconds => _isAttackPhase ? AttackMotionLockSeconds : 0f;

        public IReadOnlyList<IConsideration> Considerations { get; } = new IConsideration[]
        {
            new RoleGateConsideration(RoleFlag.Attacker, key: "DestroySpawnerGoal:IsAttacker"),
            new ColorMatchConsideration("DestroySpawnerGoal:ColorMatch"),
            new DistanceConsideration("DestroySpawnerGoal:PreferNearest", AttackerTuning.SearchRadius, preferNear: true),
            new StickyTargetConsideration(key: "DestroySpawnerGoal:Sticky"),
        };

        public IEnumerable<object> GetTargetCandidates(UtilityContext context)
        {
            // ゴール確定済み: そのスポナーがまだ有効な間は他候補を一切見ず、これだけを返し続ける
            // (=毎Thinkの再比較で対象が乗り換わらないようにする、ゴールベースAI的な固定)。
            if (_committedSpawner != null)
            {
                IEnemySpawnerInfo committed = context.Spawners.FirstOrDefault(s => ReferenceEquals(s.Transform, _committedSpawner));
                if (committed != null)
                {
                    yield return committed;
                    yield break;
                }

                _committedSpawner = null; // 破壊された/対象外になった: ゴールを解放し、通常のUtility比較に戻す
            }

            // ゴール未確定: 通常のUtility AIと同様、全スポナー候補をConsiderationsで比較させる。
            foreach (IEnemySpawnerInfo spawner in context.Spawners)
            {
                yield return spawner;
            }
        }

        public void Act(UtilityContext context, object target, NpcDecisionResult result)
        {
            if (target is not IEnemySpawnerInfo spawner || spawner.Transform == null) return;

            _committedSpawner = spawner.Transform; // ここでゴールとして確定する(既に確定済みならそのまま維持)

            float distance = context.DistanceTo(spawner.Position);
            if (distance > AttackerTuning.MeleeRange)
            {
                // フェーズ1: 接近。間合いに入るまでは攻撃せず、ひたすら近づく。
                _isAttackPhase = false;
                result.MoveDestination = SimpleObstacleAvoidance.AvoidEnemiesOnPath(context, spawner.Position);
                return;
            }

            // フェーズ2: 攻撃。間合いに入ったらその場に留まり(MoveDestinationには触れない)、
            // クールダウンが空いていれば殴り続ける。
            _isAttackPhase = true;
            if (!context.Cooldowns.IsReady(Name)) return;

            context.Cooldowns.SetCooldown(Name, CooldownDuration);
            result.Commands.Add(new WeakAttackCommand(spawner.Transform.gameObject));

            EnemySpawner spawnerComponent = spawner.Transform.GetComponent<EnemySpawner>();
            spawnerComponent?.TakeDamage(DamagePerHit);
        }
    }
}
