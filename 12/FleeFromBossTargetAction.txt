using System.Collections.Generic;
using UnityEngine;
using Game.World;

namespace Game.AI
{
      public class ConstantConsideration : ConsiderationBase
  {
      private readonly float _value;

      public ConstantConsideration(float value, string key = "Constant", float defaultWeight = 1f)
          : base(key, defaultWeight)
      {
          _value = value;
      }

      protected override float GetRawValue(UtilityContext context, object target) => _value;
  }

    public class FleeFromBossTargetAction : MoveToTargetActionBase, IMotionLockingAction
    {
        public override string Name => "FleeFromBossTarget";
        public override float BaseWeight => 5f;
        public override bool IsUrgentCapable => true;

        private const float RetreatDistance = 8f;
        private const float MotionLockDurationSeconds = 1.2f;

        public float MotionLockSeconds => MotionLockDurationSeconds;

        public override IReadOnlyList<IConsideration> Considerations { get; } = new IConsideration[]
        {
            new ConstantConsideration(1f, "FleeFromBossTarget:Baseline"),
        };

        public override IEnumerable<object> GetTargetCandidates(UtilityContext context)
        {
            if (context.Boss != null && context.Boss.IsAlive && EnemyTargetState.CurrentTarget == context.Self)
            {
                yield return context.Boss;
            }
        }

        public override void Act(UtilityContext context, object target, NpcDecisionResult result)
        {
            if (target is not IBossInfo boss) return;

            Vector3 awayDirection = context.SelfPosition - boss.Position;
            awayDirection.y = 0f;
            awayDirection = awayDirection.sqrMagnitude > 0.001f ? awayDirection.normalized : Vector3.forward;

            result.MoveDestination = context.SelfPosition + awayDirection * RetreatDistance;
        }
    }
}
