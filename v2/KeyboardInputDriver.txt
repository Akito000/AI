using System.Collections.Generic;
using UnityEngine;
using UnityEngine.InputSystem;

namespace Game.CharacterInput
{
    /// <summary>
    /// キーボード+マウス入力からコマンドを生成し、context.Commandsにセットするドライバー。
    /// Priorityを最大にしてあるので、必ず一番最初に実行される。
    /// 後続のMoveDriver/AttackDriver等は、ここでセットされたcontext.Commandsを見て動く。
    /// 移動:WASD/矢印キー, 特殊アクション:マウス左クリック, 役割変え:スペースキー。
    /// </summary>
    public class KeyboardInputDriver : CharacterDriverBase
    {
        public override int Priority => 1000;

        private readonly List<ICommand> _commands = new List<ICommand>(4);

        public override void Execute(CharacterContext context)
        {
            _commands.Clear();

            var keyboard = Keyboard.current;
            var mouse = Mouse.current;

            if (keyboard != null)
            {
                Vector2 direction = Vector2.zero;

                if (keyboard.wKey.isPressed || keyboard.upArrowKey.isPressed) direction.y += 1f;
                if (keyboard.sKey.isPressed || keyboard.downArrowKey.isPressed) direction.y -= 1f;
                if (keyboard.aKey.isPressed || keyboard.leftArrowKey.isPressed) direction.x -= 1f;
                if (keyboard.dKey.isPressed || keyboard.rightArrowKey.isPressed) direction.x += 1f;

                _commands.Add(direction.sqrMagnitude > 0.0001f
                    ? new MoveCommand(direction.normalized)
                    : IdleCommand.Instance);

                if (keyboard.spaceKey.wasPressedThisFrame)
                {
                    _commands.Add(RoleChangeCommand.Instance);
                }
            }
            else
            {
                _commands.Add(IdleCommand.Instance);
            }

            if (mouse != null && mouse.leftButton.wasPressedThisFrame)
            {
                _commands.Add(SpecialActionCommand.Instance);
            }

            context.Commands = _commands;
        }
    }
}
